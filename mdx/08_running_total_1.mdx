-- Operation: Running total (Departure Date: Month)
-- Question: What is the year-to-date ticket revenue at the end of each departure month of 2021?
-- Measure: Total Price
-- Answer key: mdx/answers/08_running_total_1.sql
-- Year-to-date revenue: SUM over PERIODSTODATE(Year level, month) (no Time-type attributes needed).
WITH MEMBER [Measures].[Value] AS
  SUM(PERIODSTODATE([Departure Date].[Calendar].[Year], [Departure Date].[Calendar].CurrentMember),
      [Measures].[Total Price])
SELECT {[Measures].[Value]} ON COLUMNS,
  NONEMPTY([Departure Date].[Calendar].[Month].Members, [Measures].[Segment Count]) ON ROWS
FROM [FlightCube]
